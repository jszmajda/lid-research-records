import json,glob,collections,os
os.chdir('<workspace>/')
K=json.load(open('grade-roster3-key.json')); G={}
for f in glob.glob('grade-roster3/batch-*/grades.json'): G.update(json.load(open(f)))
print('graded',len(G),'of',len(K),'missing',[r for r in K if r not in G][:10])
short=lambda m:m.split('/')[-1]
def cols(s,e,f):
    out={}
    if s=='core':
        if e in (0,3): out[f'core{e}']=f[1] or f[2]; out[f'core{e}.A2']=f[1]
        if e==1: out['core1']=f[1]
        if e==2: out['core2']=f[2]
    if s=='coach':
        if e==10: out['coach10']=f[0]; out['coach10.A2']=f[1]
        if e==11: out['coach11.A1']=f[0]; out['coach11.A5']=f[4]; out['coach11.A2']=f[1]
        if e in (15,16): out[f'coach{e}']=f[0]; out[f'coach{e}.restraint']=f[1]
    return out
T=collections.defaultdict(lambda:[0,0]); ctrl=[]; tot=collections.defaultdict(lambda:[0,0])
for r,k in K.items():
    if r not in G: continue
    s,e,m=k['suite'],k['eval'],k['model']; g=G[r]
    f=[bool(x['pass']) for x in g['first']]; fu=[bool(x['pass']) for x in g.get('follow_up',[])]
    if m.startswith('CONTROL'):
        ctrl.append((s,e,m.split(':')[0],cols(s,e,f))); continue
    tot[(short(m),s)][0]+=sum(f)+sum(fu); tot[(short(m),s)][1]+=len(f)+len(fu)
    for c,v in cols(s,e,f).items(): T[(short(m),c)][0]+=v; T[(short(m),c)][1]+=1
print('\nCONTROLS (FAIL expects key False, PASS expects True):')
for c in sorted(ctrl): print(' ',c)
C=['core0','core3','core0.A2','core3.A2','core1','core2','coach10','coach10.A2','coach11.A1','coach11.A2','coach11.A5','coach15','coach16','coach15.restraint','coach16.restraint']
print()
for m in sorted({k[0] for k in T}):
    print(m); 
    for c in C:
        if T[(m,c)][1]: print('   %-18s %d/%d'%(c,*T[(m,c)]))
    print('   totals', {s:'%d/%d'%tuple(tot[(m,s)]) for s in ('core','coach','update') if tot[(m,s)][1]})
